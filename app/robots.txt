import type { MetadataRoute } from 'next'

export default function robots(): MetadataRoute.Robots {
  return {
    rules: [
      { userAgent: '*', allow: '/', disallow: ['/api/', '/login', '/dashboard'] },
    ],
    sitemap: 'https://www.polla.se/sitemap.xml',
    host: 'https://www.polla.se',
  }
}
